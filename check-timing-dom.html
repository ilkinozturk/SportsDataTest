Check Timing Analytics DOM Values

🔍 Quick Check Script

Run this in console on the team-stats.html page (Goals tab):

// Check if we're on Goals tab const goalsTab = document.querySelector('.tab-button.active'); console.log('Current tab:', goalsTab?.textContent); // Check if timing section is visible const timingSection = document.getElementById('timing-analytics-section'); console.log('Timing section visible:', timingSection?.style.display !== 'none'); // Check current timing filter const activeFilter = document.querySelector('#timing-analytics-section .section-filter.active'); console.log('Active timing filter:', activeFilter?.textContent); // Check bar values const periods = ['0_15', '16_30', '31_45', '46_60', '61_75', '76_90']; console.log('\n=== TIMING BAR VALUES ==='); periods.forEach(period => { const scoredBar = document.getElementById(`goals${period}ScoredBar`); const scoredText = document.getElementById(`goals${period}ScoredText`); const concededBar = document.getElementById(`goals${period}ConcededBar`); const concededText = document.getElementById(`goals${period}ConcededText`); console.log(`\nPeriod ${period.replace('_', '-')}:`); console.log(` Scored: ${scoredText?.textContent || 'NO TEXT'} (width: ${scoredBar?.style.width || 'NO BAR'})`); console.log(` Conceded: ${concededText?.textContent || 'NO TEXT'} (width: ${concededBar?.style.width || 'NO BAR'})`); });

🐛 Force Update Test

If values show 0, run this to force update with HOME data:

// Force update timing analytics with HOME filter if (window.TeamStatsGoalsDisplay && window.globalStatistics) { // First check if HOME data exists console.log('HOME data check:', { homeGoals0_15: window.globalStatistics.homeGoals0_15, homeGoals16_30: window.globalStatistics.homeGoals16_30, homeGoals31_45: window.globalStatistics.homeGoals31_45, homeGoals46_60: window.globalStatistics.homeGoals46_60, homeGoals61_75: window.globalStatistics.homeGoals61_75, homeGoals76_90: window.globalStatistics.homeGoals76_90 }); // Update timing analytics window.TeamStatsGoalsDisplay.updateTimingAnalytics(window.globalStatistics, 'home'); // Check first bar after update setTimeout(() => { const bar = document.getElementById('goals0_15ScoredBar'); const text = document.getElementById('goals0_15ScoredText'); console.log('After update - 0-15 scored:', text?.textContent, 'width:', bar?.style.width); }, 100); } else { console.error('Required objects not found!'); }

🔧 Direct DOM Update Test

Test if DOM elements can be updated directly:

// Test direct DOM manipulation const testBar = document.getElementById('goals0_15ScoredBar'); const testText = document.getElementById('goals0_15ScoredText'); if (testBar && testText) { console.log('Before:', testBar.style.width, testText.textContent); // Set test values testBar.style.width = '50%'; testText.textContent = '4 (24%)'; console.log('After:', testBar.style.width, testText.textContent); console.log('✓ DOM elements can be updated'); } else { console.error('✗ DOM elements not found!'); }
